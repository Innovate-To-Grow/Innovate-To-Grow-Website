function submitReplyForm(form, replyArea) { var actionUrl = form.getAttribute('data-action'); if (!actionUrl) return; var sendBtn = form.querySelector('[data-inbox-send-btn]'); var statusEl = form.querySelector('[data-inbox-reply-status]'); if (sendBtn) { sendBtn.disabled = true; sendBtn.textContent = 'Sending...'; } if (statusEl) { statusEl.className = 'mt-2 text-sm hidden'; statusEl.textContent = ''; } var csrfToken = document.querySelector('[name=csrfmiddlewaretoken]'); if (!csrfToken) csrfToken = document.cookie.match(/csrftoken=([^;]+)/); var body = new FormData(form); var headers = { 'X-Requested-With': 'XMLHttpRequest' }; var csrfValue = csrfToken ? (csrfToken.value || csrfToken[1]) : ''; if (csrfValue) headers['X-CSRFToken'] = csrfValue; fetch(actionUrl, { method: 'POST', credentials: 'same-origin', headers: headers, body: body, }) .then(function (r) { return r.json(); }) .then(function (data) { if (data.ok) { showReplySent(replyArea, data.message); } else { if (statusEl) { statusEl.className = 'mt-2 text-sm text-red-600'; statusEl.textContent = data.error || 'Failed to send.'; } if (sendBtn) { sendBtn.disabled = false; sendBtn.textContent = 'Send Reply'; } } }) .catch(function () { if (statusEl) { statusEl.className = 'mt-2 text-sm text-red-600'; statusEl.textContent = 'Network error. Please try again.'; } if (sendBtn) { sendBtn.disabled = false; sendBtn.textContent = 'Send Reply'; } }); } /* Event delegation */ document.body.addEventListener('click', function (e) { var listToggle = e.target.closest('[data-inbox-list-toggle]'); if (listToggle) { e.preventDefault(); var listRoot = getRoot(listToggle); if (!listRoot) return; var listShell = listRoot.querySelector('.i2g-mail-shell'); var nextListCollapsed = listShell ? !listShell.classList.contains('is-list-collapsed') : false; setListCollapsed(listRoot, nextListCollapsed); storeListCollapsed(nextListCollapsed); resizePreviewIframe(); return; } var accountToggle = e.target.closest('[data-inbox-account-toggle]'); if (accountToggle) { e.preventDefault(); var accountRoot = getRoot(accountToggle); if (!accountRoot) return; var accountShell = accountRoot.querySelector('.i2g-mail-shell'); var nextCollapsed = accountShell ? !accountShell.classList.contains('is-account-collapsed') : false; setAccountCollapsed(accountRoot, nextCollapsed); storeAccountCollapsed(nextCollapsed); return; } var folderTab = e.target.closest('[data-inbox-folder]'); if (folderTab) { e.preventDefault(); var folderRoot = getRoot(folderTab); if (!folderRoot) return; var nextFolder = folderTab.getAttribute('data-inbox-folder') === 'SENT' ? 'SENT' : 'INBOX'; if (nextFolder !== getFolder(folderRoot)) { storeFolder(nextFolder); setActiveFolderTab(folderRoot, nextFolder); // A uid is unique only within its folder, so drop the open preview/hash. clearPreviewPane(folderRoot); setHashUid(null); fetchInbox(folderRoot, { limit: getLimit(folderRoot), folder: nextFolder, account: getAccount(folderRoot), refresh: '1' }); } return; } var btn = e.target.closest('[data-inbox-refresh]'); if (btn) { e.preventDefault(); var root = getRoot(btn); if (root) fetchInbox(root, { limit: getLimit(root), folder: getFolder(root), account: getAccount(root), refresh: '1' }); return; } // Reply link → load form into #inbox-reply-area (below email) var replyLink = e.target.closest('[data-inbox-reply-link]'); if (replyLink) { e.preventDefault(); var replyArea = document.getElementById('inbox-reply-area'); if (replyArea) { replyLink.style.display = 'none'; loadReplyForm(replyArea, replyLink.getAttribute('data-inbox-reply-link')); } return; } // Toggle Cc field var ccToggle = e.target.closest('[data-toggle-cc]'); if (ccToggle) { e.preventDefault(); var ccRow = document.querySelector('[data-cc-row]'); if (ccRow) { var visible = ccRow.style.display !== 'none'; ccRow.style.display = visible ? 'none' : 'flex'; ccToggle.style.display = visible ? '' : 'none'; if (!visible) ccRow.querySelector('input').focus(); } return; } // Cancel reply → just clear the reply area var cancelBtn = e.target.closest('[data-inbox-cancel-reply]'); if (cancelBtn) { e.preventDefault(); var replyArea = document.getElementById('inbox-reply-area'); if (replyArea) replyArea.innerHTML = ''; var replyLink = document.querySelector('[data-inbox-reply-link]'); if (replyLink) replyLink.style.display = ''; return; } var row = e.target.closest('[data-inbox-row]'); if (row) { e.preventDefault(); var root = getRoot(row); if (!root) return; var pane = root.querySelector('#inbox-preview-pane'); if (pane) { loadPreview(root, row); } else { window.location.href = row.getAttribute('href'); } } }); // Reply form submission document.body.addEventListener('submit', function (e) { var form = e.target.closest('[data-inbox-reply-form]'); if (form) { e.preventDefault(); var replyArea = document.getElementById('inbox-reply-area'); if (replyArea) submitReplyForm(form, replyArea); } }); document.body.addEventListener('change', function (e) { var accountSel = e.target.closest('[data-inbox-account]'); if (accountSel) { var accountRoot = getRoot(accountSel); if (!accountRoot) return; storeAccount(accountSel.value); // The open message belongs to the previous selection: drop the preview/hash. clearPreviewPane(accountRoot); setHashUid(null); fetchInbox(accountRoot, { limit: getLimit(accountRoot), folder: getFolder(accountRoot), account: accountSel.value }); return; } var sel = e.target.closest('[data-inbox-limit]'); if (!sel) return; var root = getRoot(sel); if (root) fetchInbox(root, { limit: getLimit(root), folder: getFolder(root), account: getAccount(root) }); }); /* Auto-load on page ready */ var root = document.getElementById('inbox-all-dynamic'); if (root) fetchInbox(root, { folder: getStoredFolder(), account: getStoredAccount() }); /* Auto-refresh: the server polls every mailbox for new mail in the background; while this page is visible it reloads its list (served from that poll, so it seldom connects to a mailbox). Skipped while a reply is being written or a load is already running. */ function autoRefreshSeconds(r) { var shell = r ? r.querySelector('[data-inbox-auto-refresh]') : null; var value = shell ? parseInt(shell.getAttribute('data-inbox-auto-refresh'), 10) : 0; return value > 0 ? value : 0; } var lastAutoRefresh = Date.now(); window.setInterval(function () { var r = document.getElementById('inbox-all-dynamic'); var seconds = autoRefreshSeconds(r); if (!r || !seconds || document.hidden) return; if (Date.now() - lastAutoRefresh < seconds * 1000) return; if (r.querySelector('#inbox-reply-area form')) return; var busyBtn = r.querySelector('[data-inbox-refresh][disabled]'); if (busyBtn) return; lastAutoRefresh = Date.now(); fetchInbox(r, { limit: getLimit(r), folder: getFolder(r), account: getAccount(r) }, { quiet: true }); }, 5000); window.addEventListener('resize', resizePreviewIframe); window.addEventListener('beforeunload', function () { var r = document.getElementById('inbox-all-dynamic'); if (r) persistScrollToSession(r); }); })();
